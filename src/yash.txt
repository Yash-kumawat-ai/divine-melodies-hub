u are working on the Raghavam devotional web app (Vite + React + TypeScript + Tailwind + Supabase).

The current `panchangTemple.ts` has a STATIC list of Shubh/Ashubh Karya that never changes day to day. Replace it with a fully dynamic engine that recomputes instantly based on today's Tithi, Vara, Nakshatra, and special time windows. The UI should feel snappy — no loading states, no API calls, pure TypeScript computation.

---

## ARCHITECTURE OVERVIEW

Create ONE new file: `src/utils/shubhKaryaEngine.ts`
This file contains ALL master data + the compute function.
Zero API calls. Zero async. Pure synchronous TypeScript.
The component that currently reads static Karya lists should call this engine instead.

## STEP 5 — THE ENGINE FUNCTION

```typescript
export interface PanchangInput {
  tithiNumber: number;        // 1–30
  varaIndex: number;          // 0=Sun … 6=Sat
  nakshatraName: string;      // Hindi name e.g. 'रोहिणी'
  isRahuKaal: boolean;
  isAbhijitMuhurat: boolean;
  isBrahmaKaal: boolean;
  isGulikKaal: boolean;
}

export interface KaryaItem {
  name: string;
  reason: string;             // badge text e.g. "रोहिणी नक्षत्र • सोमवार"
  source: 'tithi' | 'vara' | 'nakshatra' | 'special';
  priority: number;           // 1=high 2=medium 3=low — for sort order
}

export interface ShubhAshubhResult {
  shubhKarya: KaryaItem[];
  ashubhKarya: KaryaItem[];
  todaySummary: string;
  dominantFactor: string;
  tithiJati: string;
  isRiktaTithi: boolean;
  activeWarning: string | null;   // Rahu Kaal / Gulik Kaal warning if active
  activeBoon: string | null;      // Abhijit / Brahma Kaal if active
}

export function computeShubhAshubhKarya(input: PanchangInput): ShubhAshubhResult {
  const tithi = TITHI_RULES[input.tithiNumber];
  const vara = VARA_RULES[input.varaIndex];
  const nakshatra = NAKSHATRA_RULES[input.nakshatraName] ?? null;

  const varaDayName = ['रविवार','सोमवार','मंगलवार','बुधवार','गुरुवार','शुक्रवार','शनिवार'][input.varaIndex];

  // Build shubh list
  const shubhMap = new Map<string, KaryaItem>();

  // Priority 3 = Tithi (weakest)
  tithi.shubh.forEach(k => shubhMap.set(k, {
    name: k, reason: `${tithi.note.split('—')[0].trim()} तिथि`,
    source: 'tithi', priority: 3
  }));

  // Priority 2 = Vara
  vara.shubh.forEach(k => shubhMap.set(k, {
    name: k, reason: `${varaDayName} — ${vara.specialNote}`,
    source: 'vara', priority: 2
  }));

  // Priority 1 = Nakshatra (strongest, overrides)
  if (nakshatra) {
    nakshatra.shubh.forEach(k => shubhMap.set(k, {
      name: k, reason: `${input.nakshatraName} नक्षत्र — ${nakshatra.qualityEffect}`,
      source: 'nakshatra', priority: 1
    }));
  }

  // Build ashubh list (same priority order)
  const ashubhMap = new Map<string, KaryaItem>();

  tithi.ashubh.forEach(k => ashubhMap.set(k, {
    name: k, reason: `${tithi.jati} तिथि — ${tithi.note.split('—')[1]?.trim() ?? ''}`,
    source: 'tithi', priority: 3
  }));

  vara.ashubh.forEach(k => ashubhMap.set(k, {
    name: k, reason: `${varaDayName} — ${vara.rulingPlanet} प्रभाव`,
    source: 'vara', priority: 2
  }));

  if (nakshatra) {
    nakshatra.ashubh.forEach(k => ashubhMap.set(k, {
      name: k, reason: `${input.nakshatraName} नक्षत्र (${nakshatra.quality}) — ${nakshatra.qualityEffect}`,
      source: 'nakshatra', priority: 1
    }));
  }

  // CONFLICT RESOLUTION:
  // If item appears in both: if nakshatra quality is 'तीक्ष्ण' or 'उग्र' → keep in ashubh
  // else if it's in vara.shubh → keep in shubh
  // else → keep in ashubh (safer)
  for (const key of shubhMap.keys()) {
    if (ashubhMap.has(key)) {
      const isHarshNakshatra = nakshatra && ['तीक्ष्ण', 'उग्र'].includes(nakshatra.quality);
      if (isHarshNakshatra) {
        shubhMap.delete(key); // ashubh wins
      } else {
        ashubhMap.delete(key); // shubh wins
      }
    }
  }

  // SPECIAL WINDOWS override
  let activeWarning: string | null = null;
  let activeBoon: string | null = null;

  if (input.isRahuKaal) {
    activeWarning = SPECIAL_WINDOWS.rahuKaal.note;
    shubhMap.clear(); // ALL shubh work is forbidden during Rahu Kaal
    SPECIAL_WINDOWS.rahuKaal.ashubh.forEach(k => ashubhMap.set(k, {
      name: k, reason: 'राहु काल — राहु ग्रह का भ्रम और अवरोध',
      source: 'special', priority: 1
    }));
  }

  if (input.isAbhijitMuhurat) {
    activeBoon = SPECIAL_WINDOWS.abhijitMuhurat.note;
    SPECIAL_WINDOWS.abhijitMuhurat.shubh.forEach(k => shubhMap.set(k, {
      name: k, reason: 'अभिजित मुहूर्त — ब्रह्मा का समय, सर्वश्रेष्ठ',
      source: 'special', priority: 1
    }));
    // Abhijit reduces Rikta tithi's ashubh impact if active
    if (tithi.jati === 'रिक्ता') {
      for (const [key, val] of ashubhMap.entries()) {
        if (val.source === 'tithi') ashubhMap.delete(key);
      }
    }
  }

  if (input.isBrahmaKaal) {
    activeBoon = SPECIAL_WINDOWS.brahmaKaal.note;
    SPECIAL_WINDOWS.brahmaKaal.shubh.forEach(k => shubhMap.set(k, {
      name: k, reason: 'ब्रह्म मुहूर्त — प्राण वायु शुद्धतम',
      source: 'special', priority: 1
    }));
  }

  if (input.isGulikKaal && !input.isRahuKaal) {
    activeWarning = SPECIAL_WINDOWS.gulikKaal.note;
    SPECIAL_WINDOWS.gulikKaal.ashubh.forEach(k => ashubhMap.set(k, {
      name: k, reason: 'गुलिक काल — शनि पुत्र का समय',
      source: 'special', priority: 1
    }));
    SPECIAL_WINDOWS.gulikKaal.shubh.forEach(k => shubhMap.set(k, {
      name: k, reason: 'गुलिक काल में शनि उपाय अत्यंत शुभ',
      source: 'special', priority: 1
    }));
  }

  // Build todaySummary
  const dominantFactor = input.isRahuKaal
    ? 'राहु काल'
    : nakshatra?.quality === 'स्थिर' ? `${input.nakshatraName} नक्षत्र`
    : tithi.jati === 'पूर्णा' ? tithi.note.split('—')[0].trim()
    : varaDayName;

  const todaySummary = `आज ${varaDayName}, ${input.nakshatraName} नक्षत्र, ${tithi.note.split('—')[0].trim()} — ${tithi.deity} का आशीर्वाद। ${tithi.note.split('—')[1]?.trim() ?? ''}`;

  return {
    shubhKarya:    Array.from(shubhMap.values()).sort((a, b) => a.priority - b.priority),
    ashubhKarya:   Array.from(ashubhMap.values()).sort((a, b) => a.priority - b.priority),
    todaySummary,
    dominantFactor,
    tithiJati:     tithi.jati,
    isRiktaTithi:  tithi.jati === 'रिक्ता',
    activeWarning,
    activeBoon
  };
}
```

---IMPORTANT: The app already fetches full Panchang data from VedAstro via 
panchangService.ts with daily caching per city. The VedAstro response 
contains real computed timings for Rahu Kaal, Gulik Kaal, Abhijit Muhurat, 
and Brahma Kaal for the user's city. 

DO NOT hardcode any time values like "12:00" or "4:00 AM". 
ALL time window checks must use the actual timestamps from the 
VedAstro API response already stored in panchangData. 
If a particular field (e.g. abhijitStart) is absent from the VedAstro 
response, set that flag to false — never fallback to a hardcoded time.
